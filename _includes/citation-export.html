{% comment %} Renders "BibTeX" / "RIS" download buttons for one publication. The citation data is emitted as data attributes and turned into a file by the script in citation-export-script.html. Usage: {% include citation-export.html item=post %} {% endcomment %} {% include base_path %} {% assign cite = include.item %} {% case cite.category %} {% when 'books' %} {% assign cite_bibtype = 'incollection' %} {% assign cite_ristype = 'CHAP' %} {% assign cite_entype = 'Book Section' %} {% when 'conferences' %} {% assign cite_bibtype = 'inproceedings' %} {% assign cite_ristype = 'CONF' %} {% assign cite_entype = 'Conference Paper' %} {% else %} {% assign cite_bibtype = 'article' %} {% assign cite_ristype = 'JOUR' %} {% assign cite_entype = 'Journal Article' %} {% endcase %} {% if cite.permalink %} {% assign cite_key = cite.permalink | split: "/" | last %} {% else %} {% assign cite_key = cite.title | slugify %} {% endif %} {% if cite.paperurl and cite.paperurl != '' %} {% assign cite_url = cite.paperurl %} {% else %} {% assign cite_url = cite.url | prepend: base_path %} {% endif %}
